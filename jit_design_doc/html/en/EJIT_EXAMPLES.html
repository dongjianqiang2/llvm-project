EJIT Examples

Minimal code snippets for common scenarios: static, single-dimension, multi-dimension, lifecycle, async, and bare-metal symbol registration.

1. Static global config specialization

Suits board-level config and global config that does not change after compilation. No ejit_activate needed.

struct BoardConfig {
    EJIT_PERIOD_CONST uint32_t boardType;
    uint32_t flags;
};

EJIT_IN_PERIOD(static) struct BoardConfig g_boardCfg;

EJIT_ENTRY
void process_board(int x) {
    if (g_boardCfg.boardType == 1) {
        board_type_1_path(x);
    } else {
        generic_path(x);
    }
}

2. Single-dimension cell time window

struct CellConfig {
    EJIT_PERIOD_CONST uint32_t mode;
    EJIT_PERIOD_CONST uint32_t threshold;
};

EJIT_IN_PERIOD_ARRAY(cell) struct CellConfig g_cellCfg[16];

EJIT_ENTRY
int process_cell(EJIT_DIM(cell) uint8_t cellIdx, int value) {
    if (g_cellCfg[cellIdx].mode == 1)
        return value + g_cellCfg[cellIdx].threshold;
    return value;
}

void run(void) {
    g_cellCfg[2].mode = 1;
    g_cellCfg[2].threshold = 7;
    ejit_activate("cell", 2);
    process_cell(2, 100);
}

3. Multi-dimension cell + trp specialization

struct CellConfig { EJIT_PERIOD_CONST uint32_t mode; };
struct TrpConfig  { EJIT_PERIOD_CONST uint32_t type; };

EJIT_IN_PERIOD_ARRAY(cell) struct CellConfig g_cellCfg[16];
EJIT_IN_PERIOD_ARRAY(trp)  struct TrpConfig  g_trpCfg[8];

EJIT_ENTRY
void process_pair(EJIT_DIM(cell) uint8_t cellIdx,
                  EJIT_DIM(trp)  uint8_t trpIdx,
                  int payload) {
    if (g_cellCfg[cellIdx].mode == 1 && g_trpCfg[trpIdx].type == 2)
        fast_pair_path(payload);
    else
        slow_pair_path(payload);
}

Before calling, activate both time-window instances separately:

ejit_activate("cell", cellIdx);
ejit_activate("trp", trpIdx);
process_pair(cellIdx, trpIdx, payload);

4. Nested struct fields

struct Inner {
    EJIT_PERIOD_CONST uint32_t subMode;
};
struct CellConfig {
    struct Inner inner;
    uint32_t runtimeCounter;
};

EJIT_IN_PERIOD_ARRAY(cell) struct CellConfig g_cellCfg[16];

EJIT_ENTRY
void process_nested(EJIT_DIM(cell) uint8_t idx) {
    if (g_cellCfg[idx].inner.subMode == 3)
        nested_fast();
}

5. Pointer-type period array

A global pointer points to a struct array. The user must ensure it is initialized before calling EJIT_ENTRY and that indices stay in bounds.

struct CellConfig { EJIT_PERIOD_CONST uint32_t mode; };

EJIT_IN_PERIOD_ARRAY(cell) struct CellConfig *g_cellPtr;

void init_cells(struct CellConfig *buf) {
    g_cellPtr = buf;
}

EJIT_ENTRY
void process_ptr(EJIT_DIM(cell) uint8_t idx) {
    if (g_cellPtr[idx].mode == 1)
        ptr_fast();
}

6. Lifecycle function

EJIT_PERIOD_GUARD(cell)
void update_cell(EJIT_DIM(cell) uint8_t idx, uint32_t mode) {
    g_cellCfg[idx].mode = mode;
}

// equivalent manual flow:
ejit_deactivate("cell", idx);
g_cellCfg[idx].mode = mode;
ejit_activate("cell", idx);

7. Async compilation mode

ejit_config_t cfg = {0};
cfg.compileMode = EJIT_COMPILE_ASYNC;
cfg.optLevel = EJIT_OPT_L2;
ejit_init(&cfg);

ejit_activate("cell", idx);
process_cell(idx, 100); // first call usually falls back and submits a background compile
process_cell(idx, 101); // subsequent calls may hit the JIT cache
In async mode, side effects of EJIT_ENTRY should be idempotent, to avoid business divergence between the first AOT fallback and later JIT versions.

8. Bare-metal external symbol registration

extern void platform_log(const char *msg);

void register_platform_symbols(void) {
    ejit_register_symbol("platform_log", (void *)&platform_log);
}

Normally PASS1 automatically scans closure references and generates registration calls; on bare-metal or special linking scenarios you can add them manually.
