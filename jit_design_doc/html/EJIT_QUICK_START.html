1. 写一个可特化的配置结构体

把时间窗内稳定的字段标记为 EJIT_PERIOD_CONST。不要标记频繁变化字段。

#include <stdint.h>
#include "llvm/ExecutionEngine/EJIT/EJitRuntime.h"

#define MODE_FAST 1
#define MODE_SAFE 2

struct CellConfig {
    EJIT_PERIOD_CONST uint32_t mode;
    EJIT_PERIOD_CONST uint32_t threshold;
    uint32_t counter;
};

EJIT_IN_PERIOD_ARRAY(cell) struct CellConfig g_cellCfg[16];

2. 标记 JIT 入口函数

EJIT_ENTRY 标记原函数；EJIT_DIM(cell) 告诉 EJIT 哪个参数是 cell 时间窗下标。

static void fast_path(int v);
static void safe_path(int v);
static void default_path(int v);

EJIT_ENTRY
void process_cell(EJIT_DIM(cell) uint8_t cellIdx, int value) {
    if (g_cellCfg[cellIdx].mode == MODE_FAST) {
        fast_path(value + g_cellCfg[cellIdx].threshold);
    } else if (g_cellCfg[cellIdx].mode == MODE_SAFE) {
        safe_path(value);
    } else {
        default_path(value);
    }
}

3. 标记配置修改函数

修改 EJIT_PERIOD_CONST 字段时，用 EJIT_PERIOD_GUARD 包住生命周期。编译器会自动插入 deactivate/activate。

EJIT_PERIOD_GUARD(cell)
void update_cell_cfg(EJIT_DIM(cell) uint8_t cellIdx,
                     uint32_t mode,
                     uint32_t threshold) {
    g_cellCfg[cellIdx].mode = mode;
    g_cellCfg[cellIdx].threshold = threshold;
}

4. 初始化 EJIT 并激活时间窗

int main(void) {
    ejit_config_t cfg = {0};
    cfg.compileMode = EJIT_COMPILE_SYNC;
    cfg.optLevel = EJIT_OPT_L2;
    cfg.maxCodeMemory = 2 * 1024 * 1024;
    cfg.maxCacheEntries = 4096;
    cfg.enableLogger = true;

    if (ejit_init(&cfg) != EJIT_OK)
        return 1;

    g_cellCfg[3].mode = MODE_FAST;
    g_cellCfg[3].threshold = 10;
    ejit_activate("cell", 3);

    process_cell(3, 100); // 首次编译或 fallback
    process_cell(3, 101); // 命中 cache 后执行特化代码

    update_cell_cfg(3, MODE_SAFE, 20); // 旧 cache 失效
    process_cell(3, 102);              // 重新编译或 fallback

    ejit_shutdown();
    return 0;
}

6. 怎么确认 JIT 生效

检查点期望
编译 IR存在 !ejit.metadata、!ejit.may_const、@__ejit_bitcode
wrapper入口调用 ejit_compile_or_get，有 jit_dispatch/jit_fallback 路径
运行时 stats首次 miss 后 cache entries 增加，后续 hits 增加
dumpJITDir优化后 IR 中配置字段 load 被常量替换，分支被折叠

7. 最常见的 5 个错误

  1. 忘记调用 ejit_init。
  2. 忘记 ejit_activate("cell", idx)。
  3. 字段没标 EJIT_PERIOD_CONST。
  4. EJIT_ENTRY 参数没标 EJIT_DIM(cell)。
  5. 直接修改配置字段，没走 EJIT_PERIOD_GUARD 或手动 deactivate/activate。
如果任何一步失败，EJIT 的原则是返回 NULL 并 fallback 到 AOT，不应该影响程序正确性。
